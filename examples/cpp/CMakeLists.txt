cmake_minimum_required(VERSION 3.21)
project(CppPerfLabExternalWorkload LANGUAGES CXX)
enable_testing()
foreach(variant baseline candidate)
  add_executable(${variant} workload.cpp)
  target_compile_features(${variant} PRIVATE cxx_std_11)
  target_compile_options(${variant} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
  if(variant STREQUAL "candidate")
    target_compile_definitions(${variant} PRIVATE CONTIGUOUS_TRAVERSAL=1)
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    target_compile_options(${variant} PRIVATE -fno-omit-frame-pointer)
  endif()
  if(PERFLAB_SANITIZERS)
    if(PERFLAB_SANITIZER_LIB_DIR)
      target_link_directories(${variant} BEFORE PRIVATE ${PERFLAB_SANITIZER_LIB_DIR})
      set_property(TARGET ${variant} APPEND PROPERTY BUILD_RPATH ${PERFLAB_SANITIZER_LIB_DIR})
    endif()
    target_compile_options(${variant} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${variant} PRIVATE -fsanitize=address,undefined)
  endif()
  if(PERFLAB_REPORTS)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(${variant} PRIVATE -fopt-info-vec-optimized -fopt-info-vec-missed)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      target_compile_options(${variant} PRIVATE -Rpass=loop-vectorize -Rpass-missed=loop-vectorize -fsave-optimization-record)
    endif()
  endif()
  foreach(kind compute locality allocations)
    foreach(size 1 17 128)
      add_test(NAME ${variant}_${kind}_${size} COMMAND ${variant} ${kind} ${size} 3)
    endforeach()
  endforeach()
endforeach()

if(PERFLAB_GOOGLE_BENCHMARK_ROOT)
  find_package(benchmark 1.9.4 EXACT CONFIG REQUIRED PATHS ${PERFLAB_GOOGLE_BENCHMARK_ROOT}/lib64/cmake/benchmark ${PERFLAB_GOOGLE_BENCHMARK_ROOT}/lib/cmake/benchmark NO_DEFAULT_PATH)
  foreach(variant baseline candidate)
    add_executable(benchmarks_${variant} benchmark_pairs.cpp)
    target_compile_features(benchmarks_${variant} PRIVATE cxx_std_17)
    target_link_libraries(benchmarks_${variant} PRIVATE benchmark::benchmark)
    if(variant STREQUAL "candidate")
      target_compile_definitions(benchmarks_${variant} PRIVATE SORT_PAIRS=1)
    endif()
  endforeach()
endif()

foreach(variant baseline candidate)
  add_executable(pairs_${variant} duplicates.cpp)
  target_compile_features(pairs_${variant} PRIVATE cxx_std_11)
  target_compile_options(pairs_${variant} PRIVATE -Wall -Wextra -Werror)
  if(variant STREQUAL "candidate")
    target_compile_definitions(pairs_${variant} PRIVATE SORT_PAIRS=1)
  endif()
  if(PERFLAB_SANITIZERS)
    if(PERFLAB_SANITIZER_LIB_DIR)
      target_link_directories(pairs_${variant} BEFORE PRIVATE ${PERFLAB_SANITIZER_LIB_DIR})
      set_property(TARGET pairs_${variant} APPEND PROPERTY BUILD_RPATH ${PERFLAB_SANITIZER_LIB_DIR})
    endif()
    target_compile_options(pairs_${variant} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pairs_${variant} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
find_program(PERFLAB_PYTHON NAMES python3.11 REQUIRED)
foreach(variant baseline candidate)
  add_test(NAME pairs_${variant}_oracle COMMAND ${PERFLAB_PYTHON}
    ${CMAKE_CURRENT_SOURCE_DIR}/oracle_pairs.py $<TARGET_FILE:pairs_${variant}>)
endforeach()
